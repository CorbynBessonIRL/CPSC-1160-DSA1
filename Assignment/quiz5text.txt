void makeNumChar (std::array <char> &arr){
    int pos = 0;
    for (int i = 1; i < arr.size(); i++){
        if (isdigit(arr[i])){
            swap(arr[i], arr[pos]);
            pos++;
        }
    }
    return;
}

fixed code:  

void makeNumChar(char arr[], int size){
    int pos = 0;

    for (int i = 0; i < size; i++){
        if (isdigit(arr[i])){
            std::swap(arr[i], arr[pos]);
            pos++;
        }
    }
}
