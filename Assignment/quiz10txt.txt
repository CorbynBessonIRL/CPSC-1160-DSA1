            node *next;
        };

        node *head;

        node *delete_list(node *p);
};




quiz::quiz(const std::string& str) {

  head = nullptr;
  for (int i = str.size()-1; i>= 0; i--){
    head = new node{new char(str[i]), head};
  }
}



void quiz::print() const {
    node *p = head;

    while (p != nullptr) {
        std::cout << *(p->data);
        p = p->next;
    }

    std::cout << std::endl;
}
